4.2. Comparing Classification Methods

Code 
vals <- dfw_5070$adoption_rate

format_pct_breaks <- function(brks) {
  paste0(
    scales::percent(head(brks, -1), accuracy = 0.1),
    " – ",
    scales::percent(tail(brks, -1), accuracy = 0.1)
  )
}

brks_equal <- classIntervals(vals, n = 4, style = "equal")$brks
brks_quantile <- classIntervals(vals, n = 4, style = "quantile")$brks
brks_jenks <- classIntervals(vals, n = 4, style = "jenks")$brks

dfw_class <- dfw_5070 |>
  mutate(
    cls_equal = cut(
      adoption_rate,
      breaks = brks_equal,
      include.lowest = TRUE,
      labels = format_pct_breaks(brks_equal)
    ),
    cls_quantile = cut(
      adoption_rate,
      breaks = brks_quantile,
      include.lowest = TRUE,
      labels = format_pct_breaks(brks_quantile)
    ),
    cls_jenks = cut(
      adoption_rate,
      breaks = brks_jenks,
      include.lowest = TRUE,
      labels = format_pct_breaks(brks_jenks)
    )
  )

make_cls_map <- function(data, col_sym, title_txt) {

  ggplot(data) +
    geom_sf(
      aes(fill = {{ col_sym }}),
      color = "white",
      linewidth = 0.35
    ) +
    scale_fill_viridis_d(
      option = "viridis",
      name = "EV Adoption Rate",
      guide = guide_legend(
        nrow = 2,
        byrow = TRUE
      )
    ) +
    labs(title = title_txt) +
    theme_minimal(base_size = 10) +
    theme(
      plot.title = element_text(
        size = 11,
        face = "bold",
        hjust = 0.5
      ),
      axis.text.x = element_text(
        size = 7,
        angle = 30,
        hjust = 1
      ),
      axis.text.y = element_text(size = 7),
      axis.title = element_blank(),
      panel.grid.major = element_line(
        color = "grey90",
        linewidth = 0.25
      ),
      legend.position = "bottom",
      legend.title = element_text(size = 8, face = "bold"),
      legend.text = element_text(size = 7.5),
      legend.key.width = grid::unit(0.35, "cm"),
      legend.spacing.x = grid::unit(0.1, "cm"),
      plot.margin = margin(8, 10, 8, 10)
    )
}

m_equal <- make_cls_map(
  dfw_class,
  cls_equal,
  "Equal Interval"
)

m_quantile <- make_cls_map(
  dfw_class,
  cls_quantile,
  "Quantile"
)

m_jenks <- make_cls_map(
  dfw_class,
  cls_jenks,
  "Jenks Natural Breaks"
)

m_equal + m_quantile + m_jenks +
  plot_layout(ncol = 3)
